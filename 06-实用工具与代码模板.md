```python
import pandas as pd
import pyarrow.parquet as pq
from pathlib import Path
from typing import List, Optional

CURATED_DIR = Path("quantdb/data/curated")

def load_daily(ts_codes: Optional[List[str]] = None,
               start_date: str = "20150101",
               end_date: str = "20251231",
               fields: Optional[List[str]] = None) -> pd.DataFrame:
    """
    加载日线行情数据
    
    支持按股票代码和日期过滤，避免全量加载
    """
    daily_dir = CURATED_DIR / "daily"
    
    if ts_codes:
        # 只读指定股票
        dfs = []
        for code in ts_codes:
            file = daily_dir / f"{code}.parquet"
            if file.exists():
                df = pd.read_parquet(file, columns=fields)
                df = df[(df["trade_date"] >= start_date) & 
                        (df["trade_date"] <= end_date)]
                dfs.append(df)
        return pd.concat(dfs, ignore_index=True) if dfs else pd.DataFrame()
    else:
        # 全量加载（用 pyarrow 过滤更高效）
        dfs = []
        for file in sorted(daily_dir.glob("*.parquet")):
            df = pd.read_parquet(file, columns=fields)
            df = df[(df["trade_date"] >= start_date) & 
                    (df["trade_date"] <= end_date)]
            dfs.append(df)
        return pd.concat(dfs, ignore_index=True) if dfs else pd.DataFrame()


def load_financial_pit(table_name: str,
                        as_of_date: str,
                        ts_codes: Optional[List[str]] = None,
                        fields: Optional[List[str]] = None) -> pd.DataFrame:
    """
    PIT 模式加载财务数据
    
    只返回截至 as_of_date 实际已公告的最新财务数据
    
    参数：
        table_name: "fina_indicator" / "income" / "balancesheet" / "cashflow"
        as_of_date: PIT 截止日（YYYYMMDD）
        ts_codes: 可选，指定股票列表
    """
    file = CURATED_DIR / table_name / f"{table_name}.parquet"
    df = pd.read_parquet(file, columns=fields)
    
    if ts_codes:
        df = df[df["ts_code"].isin(ts_codes)]
    
    # PIT 核心逻辑：只用实际公告日 <= as_of_date 的记录
    ann_col = "f_ann_date" if "f_ann_date" in df.columns else "ann_date"
    df = df[df[ann_col] <= as_of_date]
    
    # 每只股票取最新的报告期数据
    df = df.sort_values([ann_col, "end_date"])
    df = df.groupby("ts_code").last().reset_index()
    
    return df


def load_index_weights(index_code: str, trade_date: str) -> pd.DataFrame:
    """加载指数成分股权重"""
    file = CURATED_DIR / "index_weight" / f"{index_code}.parquet"
    if not file.exists():
        # 尝试合并文件
        file = CURATED_DIR / "index_weight" / "index_weight.parquet"
    
    df = pd.read_parquet(file)
    df = df[df["index_code"] == index_code]
    
    # 找最近的权重日期
    available_dates = df["trade_date"].unique()
    closest_date = max(d for d in available_dates if d <= trade_date)
    
    return df[df["trade_date"] == closest_date]
```

```python
class TradeCalendar:
    """交易日历工具类"""
    
    def __init__(self):
        file = CURATED_DIR / "trade_calendar" / "trade_calendar.parquet"
        df = pd.read_parquet(file)
        self.trade_dates = sorted(
            df[df["is_open"] == 1]["cal_date"].tolist()
        )
        self._date_set = set(self.trade_dates)
    
    def is_trade_day(self, date: str) -> bool:
        """判断是否为交易日"""
        return date in self._date_set
    
    def shift(self, date: str, n: int) -> str:
        """
        从 date 偏移 n 个交易日
        n > 0: 向后移动
        n < 0: 向前移动
        """
        idx = self.trade_dates.index(date)
        return self.trade_dates[idx + n]
    
    def get_range(self, start: str, end: str) -> List[str]:
        """获取区间内的所有交易日"""
        return [d for d in self.trade_dates if start <= d <= end]
    
    def get_monthly_first(self, start: str, end: str) -> List[str]:
        """获取每月第一个交易日（用于月度调仓）"""
        dates = self.get_range(start, end)
        monthly_first = []
        current_month = ""
        for d in dates:
            month = d[:6]  # YYYYMM
            if month != current_month:
                monthly_first.append(d)
                current_month = month
        return monthly_first
```

```python
class FactorCalculator:
    """通用因子计算器"""
    
    def __init__(self, daily_df, basic_df=None):
        """
        daily_df: 日线行情数据（需含 ts_code, trade_date, OHLCV）
        basic_df: 每日指标数据（可选）
        """
        self.daily = daily_df.sort_values(["ts_code", "trade_date"])
        self.basic = basic_df
        
    def _group_apply(self, series, func, **kwargs):
        """按股票分组应用函数"""
        return self.daily.groupby("ts_code")[series].transform(func, **kwargs)
    
    # ==================== 动量类 ====================
    
    def momentum(self, n=20):
        """N日动量（收益率）"""
        return self._group_apply("close", lambda x: x.pct_change(n))
    
    def momentum_skip(self, n=20, skip=1):
        """N日动量，跳过最近skip天"""
        return self.daily.groupby("ts_code")["close"].transform(
            lambda x: x.shift(skip) / x.shift(n + skip) - 1
        )
    
    # ==================== 波动率类 ====================
    
    def volatility(self, n=20):
        """N日历史波动率"""
        ret = self._group_apply("close", lambda x: x.pct_change())
        return ret.groupby(self.daily["ts_code"]).transform(
            lambda x: x.rolling(n).std()
        ) * np.sqrt(252)
    
    def downside_vol(self, n=20):
        """N日下行波动率"""
        ret = self._group_apply("close", lambda x: x.pct_change())
        neg_ret = ret.where(ret < 0, 0)
        return neg_ret.groupby(self.daily["ts_code"]).transform(
            lambda x: x.rolling(n).std()
        ) * np.sqrt(252)
    
    def max_return(self, n=20):
        """过去N日最大单日涨幅"""
        ret = self._group_apply("close", lambda x: x.pct_change())
        return ret.groupby(self.daily["ts_code"]).transform(
            lambda x: x.rolling(n).max()
        )
    
    # ==================== 均线类 ====================
    
    def ma_bias(self, n=20):
        """均线偏离度"""
        ma = self._group_apply("close", lambda x: x.rolling(n).mean())
        return self.daily["close"] / ma - 1
    
    def ma_cross(self, short_n=5, long_n=20):
        """均线交叉（短期均线/长期均线 - 1）"""
        ma_short = self._group_apply("close", lambda x: x.rolling(short_n).mean())
        ma_long = self._group_apply("close", lambda x: x.rolling(long_n).mean())
        return ma_short / ma_long - 1
    
    # ==================== 量价类 ====================
    
    def volume_ma_ratio(self, n=20):
        """量比（当日成交量 / N日均量）"""
        ma_vol = self._group_apply("volume", lambda x: x.rolling(n).mean())
        return self.daily["volume"] / ma_vol
    
    def price_volume_corr(self, n=20):
        """量价相关系数"""
        return self.daily.groupby("ts_code").apply(
            lambda g: g["close"].rolling(n).corr(g["volume"])
        ).reset_index(level=0, drop=True)
    
    def amihud_illiquidity(self, n=20):
        """Amihud 非流动性指标"""
        ret = self._group_apply("close", lambda x: x.pct_change().abs())
        ratio = ret / self.daily["amount"]
        return ratio.groupby(self.daily["ts_code"]).transform(
            lambda x: x.rolling(n).mean()
        )
    
    # ==================== K线形态类 ====================
    
    def upper_shadow_ratio(self):
        """上影线比例"""
        high_close_open_max = self.daily["high"] - self.daily[["open", "close"]].max(axis=1)
        body = self.daily["high"] - self.daily["low"]
        return high_close_open_max / (body + 1e-8)
    
    def body_ratio(self):
        """实体比例"""
        body = (self.daily["close"] - self.daily["open"]).abs()
        range_ = self.daily["high"] - self.daily["low"]
        return body / (range_ + 1e-8)
    
    # ==================== 位置类 ====================
    
    def close_position_in_range(self, n=20):
        """收盘价在N日高低区间中的位置 (0~1)"""
        high_n = self._group_apply("high", lambda x: x.rolling(n).max())
        low_n = self._group_apply("low", lambda x: x.rolling(n).min())
        return (self.daily["close"] - low_n) / (high_n - low_n + 1e-8)
```

```python
class CrossSectionalProcessor:
    """截面因子预处理器"""
    
    @staticmethod
    def winsorize(series, lower=0.01, upper=0.99):
        """百分位截尾"""
        q_low = series.quantile(lower)
        q_high = series.quantile(upper)
        return series.clip(q_low, q_high)
    
    @staticmethod
    def mad_clip(series, n=5):
        """MAD 去极值"""
        median = series.median()
        mad = (series - median).abs().median()
        dm = 1.4826 * mad
        return series.clip(median - n * dm, median + n * dm)
    
    @staticmethod
    def zscore(series):
        """Z-Score 标准化"""
        return (series - series.mean()) / (series.std() + 1e-8)
    
    @staticmethod
    def rank_normalize(series):
        """排名百分位标准化 (0~1)"""
        return series.rank(pct=True)
    
    @staticmethod
    def industry_neutralize(series, industry):
        """行业中性化（去行业均值）"""
        ind_mean = series.groupby(industry).transform("mean")
        return series - ind_mean
    
    @staticmethod
    def size_neutralize(series, log_mv):
        """市值中性化（回归法）"""
        from numpy.polynomial.polynomial import polyval, polyfit
        coef = np.polyfit(log_mv.dropna(), series.dropna(), 1)
        return series - np.polyval(coef, log_mv)
    
    @classmethod
    def full_preprocess(cls, series, industry=None, log_mv=None):
        """完整预处理流水线"""
        result = cls.mad_clip(series)
        if industry is not None:
            result = cls.industry_neutralize(result, industry)
        if log_mv is not None:
            result = cls.size_neutralize(result, log_mv)
        result = cls.zscore(result)
        return result
```

```python
from scipy import stats

class ICAnalyzer:
    """IC 分析工具"""
    
    @staticmethod
    def calc_ic(factor_panel, return_panel, method="spearman"):
        """
        计算每期的截面 IC
        
        factor_panel: DataFrame(index=trade_date, columns=ts_code)
        return_panel: 同结构的未来收益面板
        """
        ic_series = {}
        
        common_dates = factor_panel.index.intersection(return_panel.index)
        
        for date in common_dates:
            f = factor_panel.loc[date].dropna()
            r = return_panel.loc[date].dropna()
            common_stocks = f.index.intersection(r.index)
            
            if len(common_stocks) < 30:
                continue
            
            if method == "spearman":
                ic, _ = stats.spearmanr(f[common_stocks], r[common_stocks])
            else:
                ic, _ = stats.pearsonr(f[common_stocks], r[common_stocks])
            
            ic_series[date] = ic
        
        return pd.Series(ic_series)
    
    @staticmethod
    def ic_summary(ic_series):
        """IC 统计汇总"""
        t_stat, p_value = stats.ttest_1samp(ic_series.dropna(), 0)
        
        return {
            "IC Mean": f"{ic_series.mean():.4f}",
            "IC Std": f"{ic_series.std():.4f}",
            "ICIR": f"{ic_series.mean() / ic_series.std():.4f}" if ic_series.std() > 0 else "N/A",
            "IC > 0 胜率": f"{(ic_series > 0).mean():.1%}",
            "|IC| > 0.02 比例": f"{(ic_series.abs() > 0.02).mean():.1%}",
            "t统计量": f"{t_stat:.2f}",
            "p值": f"{p_value:.4f}",
            "是否显著(p<0.05)": "✅" if p_value < 0.05 else "❌",
            "偏度": f"{ic_series.skew():.2f}",
        }
    
    @staticmethod
    def ic_decay(factor_panel, daily_returns, max_lag=60):
        """
        IC 衰减分析：计算不同滞后期的 IC
        
        看因子对未来第1天、第2天...第N天收益的预测能力
        """
        decay_curve = {}
        
        for lag in range(1, max_lag + 1, max(1, lag // 10)):
            # 计算 T+lag 的收益率
            future_return = daily_returns.shift(-lag)
            ic_series = ICAnalyzer.calc_ic(factor_panel, future_return)
            decay_curve[lag] = ic_series.mean()
        
        return pd.Series(decay_curve)
```

### 3.2 分组回测工具

```python
class GroupBacktester:
    """分组回测工具"""
    
    @staticmethod
    def group_returns(factor_panel, return_panel, n_groups=5):
        """
        按因子值分组，计算各组收益
        
        返回：DataFrame(index=trade_date, columns=[1,2,...,n_groups])
        """
        group_ret = {g: [] for g in range(1, n_groups + 1)}
        dates = []
        
        common_dates = factor_panel.index.intersection(return_panel.index)
        
        for date in common_dates:
            f = factor_panel.loc[date].dropna()
            r = return_panel.loc[date].dropna()
            common = f.index.intersection(r.index)
            
            if len(common) < n_groups * 10:
                continue
            
            dates.append(date)
            ranks = f[common].rank(pct=True)
            
            for g in range(1, n_groups + 1):
                lower = (g - 1) / n_groups
                upper = g / n_groups
                if g == n_groups:
                    mask = (ranks > lower) & (ranks <= upper)
                else:
                    mask = (ranks > lower) & (ranks <= upper)
                
                if mask.sum() > 0:
                    group_ret[g].append(r[common][mask].mean())
                else:
                    group_ret[g].append(0)
        
        return pd.DataFrame(group_ret, index=dates)
    
    @staticmethod
    def group_nav(group_returns_df, cost=0.0):
        """
        计算各组累计净值
        
        cost: 调仓成本（单边）
        """
        nav = (1 + group_returns_df - cost).cumprod()
        return nav
    
    @staticmethod
    def long_short_performance(group_returns_df, rf=0.02):
        """多空组合绩效"""
        n = group_returns_df.shape[1]
        ls_returns = group_returns_df[n] - group_returns_df[1]  # Top - Bottom
        
        n_years = len(ls_returns) / 252
        total_return = (1 + ls_returns).prod() - 1
        ann_return = (1 + total_return) ** (1/n_years) - 1
        ann_vol = ls_returns.std() * np.sqrt(252)
        sharpe = (ann_return - rf) / ann_vol if ann_vol > 0 else 0
        
        nav = (1 + ls_returns).cumprod()
        max_dd = (nav / nav.cummax() - 1).min()
        
        return {
            "多空年化收益": f"{ann_return:.2%}",
            "多空年化波动": f"{ann_vol:.2%}",
            "多空夏普比率": f"{sharpe:.2f}",
            "多空最大回撤": f"{max_dd:.2%}",
        }
```

```python
class PerformanceAnalyzer:
    """回测绩效分析工具"""
    
    @staticmethod
    def calc_metrics(nav_series, benchmark_nav=None, rf=0.02):
        """计算全套绩效指标"""
        returns = nav_series.pct_change().dropna()
        n_years = len(returns) / 252
        
        total_return = nav_series.iloc[-1] / nav_series.iloc[0] - 1
        ann_return = (1 + total_return) ** (1/n_years) - 1
        ann_vol = returns.std() * np.sqrt(252)
        sharpe = (ann_return - rf) / ann_vol if ann_vol > 0 else 0
        
        # 回撤
        cummax = nav_series.cummax()
        drawdown = (nav_series - cummax) / cummax
        max_dd = drawdown.min()
        calmar = ann_return / abs(max_dd) if max_dd != 0 else 0
        
        # 胜率
        win_rate = (returns > 0).mean()
        
        result = {
            "总收益": f"{total_return:.2%}",
            "年化收益": f"{ann_return:.2%}",
            "年化波动": f"{ann_vol:.2%}",
            "夏普比率": f"{sharpe:.2f}",
            "最大回撤": f"{max_dd:.2%}",
            "卡玛比率": f"{calmar:.2f}",
            "日胜率": f"{win_rate:.2%}",
        }
        
        if benchmark_nav is not None:
            bm_returns = benchmark_nav.pct_change().dropna()
            # 对齐日期
            common = returns.index.intersection(bm_returns.index)
            excess = returns[common] - bm_returns[common]
            
            result["超额年化"] = f"{excess.mean() * 252:.2%}"
            result["跟踪误差"] = f"{excess.std() * np.sqrt(252):.2%}"
            ir = excess.mean() / excess.std() * np.sqrt(252) if excess.std() > 0 else 0
            result["信息比率"] = f"{ir:.2f}"
        
        return result
    
    @staticmethod
    def monthly_returns_table(nav_series):
        """月度收益率表（热力图数据）"""
        returns = nav_series.resample('M').last().pct_change()
        returns.index = returns.index.to_period('M')
        
        # 转为 年×月 的矩阵
        table = returns.groupby([
            returns.index.year, returns.index.month
        ]).first().unstack()
        
        return table
    
    @staticmethod
    def drawdown_analysis(nav_series, top_n=5):
        """最大回撤分析（Top-N 回撤事件）"""
        cummax = nav_series.cummax()
        drawdown = (nav_series - cummax) / cummax
        
        events = []
        in_dd = False
        dd_start = None
        
        for date, dd in drawdown.items():
            if dd < 0 and not in_dd:
                in_dd = True
                dd_start = date
            elif dd == 0 and in_dd:
                in_dd = False
                dd_period = drawdown[dd_start:date]
                events.append({
                    "开始日": dd_start,
                    "最低点": dd_period.idxmin(),
                    "恢复日": date,
                    "最大回撤": dd_period.min(),
                    "持续天数": (date - dd_start).days,
                })
        
        events_df = pd.DataFrame(events)
        return events_df.nsmallest(top_n, "最大回撤")
```

## 6. 数据质量检查工具

```python
class DataQualityChecker:
    """数据质量检查工具"""
    
    @staticmethod
    def check_daily(daily_df):
        """日线数据质量检查"""
        checks = {}
        
        # 1. 缺失值比例
        missing = daily_df.isnull().mean()
        checks["缺失值比例"] = missing[missing > 0].to_dict()
        
        # 2. 价格异常（负价格、极端涨跌幅）
        checks["负价格数量"] = (daily_df["close"] <= 0).sum()
        checks["涨幅>30%数量"] = (daily_df["pct_chg"] > 30).sum()
        checks["跌幅>30%数量"] = (daily_df["pct_chg"] < -30).sum()
        
        # 3. 日期连续性
        for code, group in daily_df.groupby("ts_code"):
            dates = pd.to_datetime(group["trade_date"])
            gaps = dates.diff().dt.days
            large_gaps = gaps[gaps > 10]  # 超过10天的断档
            if len(large_gaps) > 0:
                if "日期断档" not in checks:
                    checks["日期断档"] = {}
                checks["日期断档"][code] = len(large_gaps)
        
        # 4. 复权因子检查
        if "adj_factor" in daily_df.columns:
            neg_adj = (daily_df["adj_factor"] <= 0).sum()
            checks["异常复权因子"] = neg_adj
        
        return checks
    
    @staticmethod
    def check_financial(fina_df):
        """财务数据质量检查"""
        checks = {}
        
        # 1. PIT 字段完整性
        if "ann_date" in fina_df.columns:
            checks["ann_date缺失比例"] = fina_df["ann_date"].isnull().mean()
        
        # 2. 报告期覆盖
        checks["报告期分布"] = fina_df["end_date"].value_counts().head(10).to_dict()
        
        # 3. ROE 异常值
        if "roe" in fina_df.columns:
            checks["ROE > 100% 数量"] = (fina_df["roe"].abs() > 100).sum()
        
        return checks
```

## 8. 实验记录模板

```python
class ExperimentLogger:
    """实验记录工具"""
    
    def __init__(self, experiment_name, save_dir="experiments"):
        self.name = experiment_name
        self.save_dir = Path(save_dir) / experiment_name
        self.save_dir.mkdir(parents=True, exist_ok=True)
        self.log = {
            "experiment_name": experiment_name,
            "created_at": pd.Timestamp.now().isoformat(),
            "parameters": {},
            "results": {},
            "notes": [],
        }
    
    def log_params(self, **kwargs):
        """记录实验参数"""
        self.log["parameters"].update(kwargs)
    
    def log_result(self, **kwargs):
        """记录实验结果"""
        self.log["results"].update(kwargs)
    
    def add_note(self, note):
        """添加备注"""
        self.log["notes"].append({
            "time": pd.Timestamp.now().isoformat(),
            "note": note
        })
    
    def save(self):
        """保存实验记录"""
        import json
        with open(self.save_dir / "experiment.json", "w") as f:
            json.dump(self.log, f, indent=2, ensure_ascii=False, default=str)
    
    def save_figure(self, fig, name):
        """保存图表"""
        fig.savefig(self.save_dir / f"{name}.png", dpi=150, bbox_inches="tight")

# 使用示例
logger = ExperimentLogger("momentum_factor_v1")
logger.log_params(
    factor="20d_momentum",
    universe="全A股",
    period="2015-2025",
    preprocessing="MAD+ZScore+行业中性",
)
logger.log_result(
    ic_mean=0.032,
    icir=0.45,
    long_short_annual_return=0.12,
)
logger.add_note("20日动量在2017-2020年效果较差，可能需要做牛熊分段")
logger.save()
```
